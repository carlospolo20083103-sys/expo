// App.js
import React, { useState, useEffect } from "react";
import {
  View,
  Text,
  TouchableOpacity,
  StyleSheet,
  FlatList,
  SafeAreaView,
  Alert,
} from "react-native";
import AsyncStorage from "@react-native-async-storage/async-storage";

export default function App() {
  const [score, setScore] = useState(0);
  const [currentQuestion, setCurrentQuestion] = useState(0);
  const [showAnswer, setShowAnswer] = useState(false);
  const [answeredCorrect, setAnsweredCorrect] = useState(false);

  // 🔹 Cargar datos guardados al iniciar
  useEffect(() => {
    const loadData = async () => {
      try {
        const savedScore = await AsyncStorage.getItem("score");
        if (savedScore !== null) {
          setScore(parseInt(savedScore));
        }
      } catch (error) {
        console.log("Error cargando datos:", error);
      }
    };
    loadData();
  }, []);

  // 🔹 Guardar puntaje cada vez que cambie
  useEffect(() => {
    const saveData = async () => {
      try {
        await AsyncStorage.setItem("score", score.toString());
      } catch (error) {
        console.log("Error guardando datos:", error);
      }
    };
    saveData();
  }, [score]);

  const questions = [
    {
      question: "¿Cuál de estas acciones ayuda más a reducir el cambio climático?",
      options: [
        "Reciclar plásticos",
        "Plantar árboles",
        "Tirar basura en tachos",
        "Usar bolsas de plástico",
      ],
      answer: "Plantar árboles",
      tip: "Los árboles absorben CO₂ y ayudan a mitigar el calentamiento global.",
      recomendacion: "Intenta sembrar al menos un árbol al año en tu comunidad.",
    },
    {
      question: "¿Qué significa el símbolo ♻?",
      options: ["Desecho", "Reutilizar", "Reciclar", "Reducir"],
      answer: "Reciclar",
      tip: "El símbolo de tres flechas en ciclo representa el reciclaje.",
      recomendacion:
        "Clasifica tus residuos en casa para que puedan reciclarse más fácilmente.",
    },
    {
      question: "¿Cuál es la mejor manera de ahorrar agua en casa?",
      options: [
        "Bañarse rápido",
        "Usar manguera",
        "Dejar el grifo abierto",
        "Regar al mediodía",
      ],
      answer: "Bañarse rápido",
      tip: "Reducir el tiempo en la ducha puede ahorrar decenas de litros de agua al día.",
      recomendacion:
        "Pon un reloj en tu baño para controlar el tiempo de ducha (5 min máx).",
    },
  ];

  // 🔹 Determinar el nivel según el puntaje
  const getLevel = (points) => {
    if (points >= 120) return "🌍 Héroe del Planeta";
    if (points >= 80) return "🐾 Defensor de la Biodiversidad";
    if (points >= 40) return "💧 Guardián del Agua";
    return "🌿 Explorador Verde";
  };

  const handleAnswer = (option) => {
    if (!showAnswer) {
      if (option === questions[currentQuestion].answer) {
        setScore(score + 10);
        setAnsweredCorrect(true);
      } else {
        setAnsweredCorrect(false);
      }
      setShowAnswer(true);
    }
  };

  const nextQuestion = () => {
    setShowAnswer(false);
    if (currentQuestion < questions.length - 1) {
      setCurrentQuestion(currentQuestion + 1);
    } else {
      Alert.alert(
        "🎉 ¡Juego terminado!",
        Tu puntaje: ${score}\nNivel alcanzado: ${getLevel(score)}
      );
      setCurrentQuestion(0);
    }
  };

  const resetGame = async () => {
    setScore(0);
    setCurrentQuestion(0);
    try {
      await AsyncStorage.removeItem("score");
    } catch (error) {
      console.log("Error reseteando datos:", error);
    }
  };

  return (
    <SafeAreaView style={styles.container}>
      <Text style={styles.title}>🌱 Estrategias Ecosustentables</Text>
      <Text style={styles.score}>Puntaje: {score}</Text>
      <Text style={styles.level}>Nivel actual: {getLevel(score)}</Text>

      <View style={styles.questionContainer}>
        <Text style={styles.question}>
          {questions[currentQuestion].question}
        </Text>

        <FlatList
          data={questions[currentQuestion].options}
          keyExtractor={(item, index) => index.toString()}
          renderItem={({ item }) => (
            <TouchableOpacity
              style={[
                styles.optionButton,
                showAnswer && item === questions[currentQuestion].answer
                  ? styles.correct
                  : null,
                showAnswer && item !== questions[currentQuestion].answer
                  ? styles.incorrect
                  : null,
              ]}
              onPress={() => handleAnswer(item)}
            >
              <Text style={styles.optionText}>{item}</Text>
            </TouchableOpacity>
          )}
        />

        {showAnswer && (
          <View style={styles.explanationBox}>
            <Text style={styles.explanation}>
              ✅ Respuesta: {questions[currentQuestion].answer}
            </Text>
            <Text style={styles.tip}>💡 {questions[currentQuestion].tip}</Text>
            {answeredCorrect && (
              <Text style={styles.recommendacion}>
                🌍 Recomendación: {questions[currentQuestion].recomendacion}
              </Text>
            )}
            <TouchableOpacity style={styles.nextButton} onPress={nextQuestion}>
              <Text style={styles.nextText}>Siguiente ➡</Text>
            </TouchableOpacity>
          </View>
        )}
      </View>

      <TouchableOpacity style={styles.resetButton} onPress={resetGame}>
        <Text style={styles.resetText}>🔄 Reiniciar progreso</Text>
      </TouchableOpacity>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: "#eafaf1", padding: 20 },
  title: {
    fontSize: 24,
    fontWeight: "bold",
    textAlign: "center",
    marginBottom: 10,
    color: "#2e7d32",
  },
  score: { fontSize: 18, textAlign: "center", color: "#388e3c" },
  level: {
    fontSize: 16,
    textAlign: "center",
    marginBottom: 20,
    fontWeight: "600",
    color: "#1b5e20",
  },
  questionContainer: { flex: 1, justifyContent: "center" },
  question: {
    fontSize: 20,
    fontWeight: "600",
    marginBottom: 20,
    textAlign: "center",
    color: "#1b5e20",
  },
  optionButton: {
    backgroundColor: "#a5d6a7",
    padding: 15,
    marginVertical: 8,
    borderRadius: 10,
  },
  optionText: { fontSize: 18, textAlign: "center", color: "#1b5e20" },
  correct: { backgroundColor: "#66bb6a" },
  incorrect: { backgroundColor: "#ef9a9a" },
  explanationBox: {
    marginTop: 20,
    padding: 15,
    backgroundColor: "#c8e6c9",
    borderRadius: 10,
  },
  explanation: { fontSize: 16, fontWeight: "bold", marginBottom: 10 },
  tip: { fontSize: 14, marginBottom: 10 },
  recomendacion: {
    fontSize: 14,
    fontStyle: "italic",
    marginBottom: 10,
    color: "#1b5e20",
  },
  nextButton: { backgroundColor: "#388e3c", padding: 12, borderRadius: 8 },
  nextText: { color: "white", textAlign: "center", fontSize: 16 },
  resetButton: {
    backgroundColor: "#ef5350",
    padding: 12,
    borderRadius: 8,
    marginTop: 20,
  },
  resetText: { color: "white", textAlign: "center", fontSize: 16 },
});
